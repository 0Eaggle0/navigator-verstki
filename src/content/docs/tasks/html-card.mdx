warmup:
  task: "В поле HTML замени ___ на «Кошка Мурка». Затем нажми «Проверить»."
  html: |
    <article class="card">
      <img src="img/murka.svg" width="200" alt="___">
      <h3>Мурка</h3>
      <p>1 год, спокойная</p>
      <a href="#">Подробнее</a>
    </article>
  css: |
    .card { max-width: 220px; }
  answer:
    field: html
    missing: "Кошка Мурка"
  normal:
    task: Добавь картинке описание «Кошка Мурка». Остальная карточка готова.
    explanation: Атрибут alt задаёт описание картинки. Текст пишут между кавычками после alt=.
    example: <img src="dog.svg" alt="Пёс Шарик">
    replacement: ""
  hard:
    task: Замени div на article, «Заголовок» сделай h3, текст — p, картинке добавь alt «Кошка Мурка», а «Подробнее» сделай ссылкой a с href.
    html: |
      <div class="card">
        <img src="img/murka.svg" width="200">
        <div>Мурка</div>
        <div>1 год, спокойная</div>
        <span>Подробнее</span>
      </div>
    css: |
      .card { max-width: 220px; }
  checks:
    - exists: "article.card"
      fail: "Карточка должна быть тегом article с классом card"
    - selector: ".card h3"
      text: "Мурка"
      fail: "Кличка должна быть в заголовке h3"
    - exists: ".card p"
      fail: "Описание должно быть абзацем p"
    - js: "return doc.querySelector('.card img')?.getAttribute('alt')?.trim() === 'Кошка Мурка'"
      fail: "Описание картинки должно назвать Мурку."
    - selector: ".card a"
      attr: "href"
      fail: "«Подробнее» должно быть ссылкой a с href"
