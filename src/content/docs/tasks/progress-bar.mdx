  answer:
    field: html
    missing: "8400"
  checks:
    - js: |
        const bar = doc.querySelector('.goal progress');
        if (bar?.value !== 8400 || bar.max !== 12000) return false;
        const style = getComputedStyle(bar);
        return bar.getClientRects().length > 0 && style.visibility === 'visible' && Number(style.opacity) > 0;
      fail: "У видимой полоски должны быть текущая и максимальная суммы."
    - js: |
        const bar = doc.querySelector('.goal progress');
        const title = doc.querySelector('.goal h2');
        return !!title?.id && bar?.getAttribute('aria-labelledby') === title.id;
      fail: "Свяжи полоску с заголовком карточки."

Нативный элемент `progress` показывает долю выполненного. Например, `<progress value="3" max="5" aria-label="Прочитано глав"></progress>` показывает три главы из пяти.
