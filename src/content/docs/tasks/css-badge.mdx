warmup:
  task: "В поле CSS замени ___ на «999px». Затем нажми «Проверить»."
  html: |
    <p>Концерт в парке <span class="badge">Бесплатно</span></p>
  css: |
    :root {

    }

    .badge {

    }

    :root {
      --accent: #16A34A;
    }

    .badge {
      padding: 4px 12px;
      border-radius: ___;
      background: var(--accent);
      color: #FFFFFF;
      font-size: 14px;
    }
  answer:
    field: css
    missing: "999px"
  checks:
    - js: "return getComputedStyle(doc.documentElement).getPropertyValue('--accent').trim().toLowerCase() === '#16a34a'"
      fail: "В :root нет переменной --accent: #16A34A"
    - js: "const badge = doc.querySelector('.badge'), root = doc.documentElement; const before = getComputedStyle(badge).backgroundColor; root.style.setProperty('--accent', '#123456', 'important'); const after = getComputedStyle(badge).backgroundColor; root.style.removeProperty('--accent'); return before !== after && after === 'rgb(18, 52, 86)'"
      fail: "Фон бейджа должен меняться при изменении переменной акцента."
    - selector: ".badge"
      style: { background-color: "rgb(22, 163, 74)", color: "rgb(255, 255, 255)" }
      fail: "У .badge нужен фон --accent и белый текст"
    - selector: ".badge"
      style: { padding-top: "4px", padding-left: "12px", font-size: "14px" }
      fail: "Нужны padding: 4px 12px и font-size: 14px"
    - js: "return parseFloat(getComputedStyle(doc.querySelector('.badge')).borderTopLeftRadius) >= 100"
      fail: "Нужно border-radius: 999px — «таблетка»"
