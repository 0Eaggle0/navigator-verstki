warmup:
  task: "В поле CSS замени ___ на «center». Затем нажми «Проверить»."
  html: |
    <div class="box">
      <div class="dot"></div>
    </div>
  css: |
    .box {
      height: 200px;
      background: #EEF2FF;
    }

    .dot {
      width: 60px;
      height: 60px;
      border-radius: 50%;
      background: #4F46E5;
    }

    .box {
      display: flex;
      justify-content: ___;
      align-items: center;
    }
  answer:
    field: css
    missing: "center"
  normal:
    task: Передвинь кружок в центр по горизонтали. По вертикали он уже отцентрирован.
    explanation: У flex-строки justify-content управляет положением по горизонтали. Значение center означает центр.
    example: '.toolbar { display: flex; justify-content: center; }'
    replacement: "flex-start"
  hard:
    task: Поставь кружок .dot ровно в центр .box по горизонтали и по вертикали с помощью Flexbox.
    html: |
      <div class="box">
        <div class="dot"></div>
      </div>
    css: |
      .box {
        height: 200px;
        background: #EEF2FF;
      }

      .dot {
        width: 60px;
        height: 60px;
        border-radius: 50%;
        background: #4F46E5;
      }
  checks:
    - selector: ".box"
      style: { display: "flex" }
      fail: "У .box нет display: flex"
    - js: "const b = doc.querySelector('.box').getBoundingClientRect(), d = doc.querySelector('.dot').getBoundingClientRect(); return Math.abs((b.left + b.right) / 2 - (d.left + d.right) / 2) < 2 && Math.abs((b.top + b.bottom) / 2 - (d.top + d.bottom) / 2) < 2"
      fail: "Кружок пока не в центре. Нужны justify-content: center и align-items: center"
    - js: "return doc.querySelector('.dot').checkVisibility({ visibilityProperty: true, opacityProperty: true })"
      fail: "Кружок должен оставаться видимым."
